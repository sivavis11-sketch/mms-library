function doGet(e) {
  const p = e && e.parameter ? e.parameter : {};

  // Cross-origin PWA bridge. The bridge itself is hosted by Apps Script,
  // so it can safely use google.script.run without relying on JSONP redirects.
  if (p.bridge === '1') {
    return createPwaBridge_();
  }

  if (p.api === '1') {
    return handleApiGet_(e);
  }

  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle(APP.NAME)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');
}

function createPwaBridge_() {
  const html = `<!doctype html>
<html><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"></head>
<body><script>
(function () {
  var SOURCE = 'mms-library-bridge';

  function reply(event, message) {
    var target = event && event.origin ? event.origin : '*';
    try { event.source.postMessage(message, target); } catch (err) {}
  }

  window.addEventListener('message', function (event) {
    var data = event && event.data;
    if (!data || data.source !== SOURCE || !data.token) return;

    var action = String(data.action || '');
    var params = data.params || {};
    var token = String(data.token);

    if (!action) {
      reply(event, {source:SOURCE, token:token, ok:false, error:'Missing API action'});
      return;
    }

    try {
      google.script.run
        .withSuccessHandler(function (result) {
          reply(event, {source:SOURCE, token:token, ok:true, data:result});
        })
        .withFailureHandler(function (err) {
          reply(event, {source:SOURCE, token:token, ok:false, error:String(err && err.message || err || 'Apps Script request failed')});
        })
        .libraryApiGet(action, params);
    } catch (err) {
      reply(event, {source:SOURCE, token:token, ok:false, error:String(err && err.message || err)});
    }
  });

  window.parent.postMessage({source:SOURCE, ready:true}, '*');
})();
</script></body></html>`;

  return HtmlService.createHtmlOutput(html)
    .setTitle('MMS Library API Bridge')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');
}

function handleApiGet_(e) {
  const p = e.parameter || {};
  const callback = String(p.callback || '');

  if (!/^[A-Za-z_$][0-9A-Za-z_$]*$/.test(callback)) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:'Invalid callback'}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  try {
    let data;

    switch (String(p.action || '')) {
      case 'dashboard':
        data = apiDashboard_(p.date || today_());
        break;
      case 'weeklyTimetable':
        data = apiWeeklyTimetable_(p.date || today_());
        break;
      case 'sections':
        data = apiSections_(p.grade || '');
        break;
      case 'students':
        data = getStudents({
          search:p.search || '',
          grade:p.grade || '',
          section:p.section || ''
        });
        break;
      case 'studentProfile':
        data = getStudentProfile(p.admissionNumber || '');
        break;
      case 'attendanceForClass':
        data = getAttendanceForClass(
          p.dateStr || today_(),
          p.grade || '',
          p.section || ''
        );
        break;
      case 'reports':
        data = getReports(p.month || today_().slice(0,7));
        break;
      case 'saveAttendance':
        data = saveAttendance(JSON.parse(p.payload || '{}'));
        break;
      case 'saveReadingAssessment':
        data = saveReadingAssessment(JSON.parse(p.payload || '{}'));
        break;
      default:
        throw new Error('Unknown API action: ' + String(p.action || ''));
    }

    return jsonp_(callback, {ok:true,data:data});

  } catch (err) {
    return jsonp_(callback, {
      ok:false,
      error:String(err && err.message || err)
    });
  }
}

function jsonp_(callback, payload) {
  return ContentService
    .createTextOutput(callback + '(' + JSON.stringify(payload) + ')')
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}

function apiSections_(grade) {
  const seen = {};

  getStudents({grade:String(grade)}).forEach(s => {
    if (s.section) seen[String(s.section)] = true;
  });

  return Object.keys(seen).sort();
}

function apiTime_(value) {
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return Utilities.formatDate(
      value,
      Session.getScriptTimeZone() || 'Asia/Kolkata',
      'HH:mm'
    );
  }

  const text = String(value || '').trim();
  const m = text.match(/^(\d{1,2}):(\d{2})/);

  return m
    ? String(m[1]).padStart(2,'0') + ':' + m[2]
    : text;
}

function apiTimetable_(rows) {
  return rows.map(t => ({
    day:String(t.day || ''),
    date:t.date || '',
    start:apiTime_(t.start),
    end:apiTime_(t.end),
    grade:String(t.grade || ''),
    section:String(t.section || ''),
    classSection:String(
      t.classSection || ('Grade ' + t.grade + ' - ' + t.section)
    ),
    note:String(t.note || '')
  }));
}

function apiDashboard_(dateStr) {
  const d = getDashboard(dateStr);
  d.timetable = apiTimetable_(d.timetable || []);
  return d;
}

function apiWeeklyTimetable_(dateStr) {
  const base = new Date(dateStr + 'T12:00:00');
  const day = base.getDay();
  const mondayOffset = day === 0 ? -6 : 1 - day;
  const out = [];

  for (let i=0; i<7; i++) {
    const d = new Date(base);
    d.setDate(base.getDate() + mondayOffset + i);

    const ds = Utilities.formatDate(
      d,
      Session.getScriptTimeZone() || 'Asia/Kolkata',
      'yyyy-MM-dd'
    );

    out.push({
      date:ds,
      day:dayName_(ds),
      classes:apiTimetable_(getTimetable(ds))
    });
  }

  return out;
}
